function quickSort(arr, L, R) {
  if (L >= R) return;
  p = getMedian(arr,L,R);
  swap(arr, pIdx, R);
  let i=L, j=R-1;
  while (i < j) {
    while (arr[i] < p) i++;
    while (i < j && arr[j] > p) j--;
    if (i < j) swap(i, j);
  }
  swap(arr, i, R);
  quickSort(L, i-1);
  quickSort(i+1, R);
}
